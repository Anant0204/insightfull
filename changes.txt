const prisma = require('../config/db');

/**
 * Central authorization check for accessing another employee's data.
 * - Agent tokens may only access their own employee record.
 * - JWT users may access their own data; ADMIN/MANAGER may access
 *   employees within the same organization only.
 * Roles are always re-read from the DB so stale tokens cannot escalate.
 *
 * @param {Object} req - Express request (requires req.user from auth middleware)
 * @param {string} targetEmployeeId - Employee whose data is being accessed
 * @returns {Promise<boolean>}
 */
const canAccessEmployeeData = async (req, targetEmployeeId) => {
    const requester = req.user || {};
    if (!targetEmployeeId) return false;

    if (requester.role === 'AGENT') {
        return requester.employeeId === targetEmployeeId;
    }

    let requesterEmployeeId = requester.employeeId;
    if (!requesterEmployeeId && requester.userId) {
        const user = await prisma.user.findUnique({
            where: { id: requester.userId },
            select: { employeeId: true }
        });
        requesterEmployeeId = user?.employeeId;
    }
    if (!requesterEmployeeId) return false;
    if (requesterEmployeeId === targetEmployeeId) return true;

    const [requesterEmp, targetEmp] = await Promise.all([
        prisma.employee.findUnique({
            where: { id: requesterEmployeeId },
            select: { role: true, organizationId: true }
        }),
        prisma.employee.findUnique({
            where: { id: targetEmployeeId },
            select: { organizationId: true }
        })
    ]);

    if (!requesterEmp || !targetEmp) return false;
    if (!targetEmp.organizationId || targetEmp.organizationId !== requesterEmp.organizationId) return false;

    return requesterEmp.role === 'ADMIN' || requesterEmp.role === 'MANAGER';
};

module.exports = { canAccessEmployeeData };

const crypto = require('crypto');
const { JWT_SECRET } = require('../config/env');

const DEFAULT_TTL_SECONDS = 60 * 60; // 1 hour

const computeSignature = (canonicalPath, expires) =>
    crypto.createHmac('sha256', JWT_SECRET).update(`${canonicalPath}:${expires}`).digest('hex');

/**
 * Sign a local /uploads/... URL with a short-lived HMAC signature so private
 * files can be served to <img>/<video> tags without exposing them publicly.
 * Non-local URLs (e.g. Cloudinary) are returned unchanged.
 */
const signLocalUrl = (url, ttlSeconds = DEFAULT_TTL_SECONDS) => {
    if (typeof url !== 'string' || !url.startsWith('/uploads/')) return url;
    const expires = Math.floor(Date.now() / 1000) + ttlSeconds;
    const sig = computeSignature(url, expires);
    return `${url}?exp=${expires}&sig=${sig}`;
};

/**
 * Verify a signature generated by signLocalUrl for the given canonical path.
 */
const verifyFileSignature = (canonicalPath, expires, sig) => {
    if (!canonicalPath || !expires || !sig || typeof sig !== 'string') return false;
    const expNum = parseInt(expires, 10);
    if (!Number.isFinite(expNum) || expNum < Math.floor(Date.now() / 1000)) return false;
    const expected = computeSignature(canonicalPath, expires);
    const sigBuffer = Buffer.from(sig);
    const expectedBuffer = Buffer.from(expected);
    if (sigBuffer.length !== expectedBuffer.length) return false;
    return crypto.timingSafeEqual(sigBuffer, expectedBuffer);
};

/**
 * Convenience: sign imageUrl/videoUrl fields on a record (or array of records)
 * in place, leaving remote URLs untouched.
 */
const signRecordUrls = (record) => {
    if (!record || typeof record !== 'object') return record;
    if (record.imageUrl) record.imageUrl = signLocalUrl(record.imageUrl);
    if (record.screenshotUrl) record.screenshotUrl = signLocalUrl(record.screenshotUrl);
    if (record.videoUrl) record.videoUrl = signLocalUrl(record.videoUrl);
    return record;
};

const signRecordList = (list) => (Array.isArray(list) ? list.map(signRecordUrls) : list);

module.exports = { signLocalUrl, verifyFileSignature, signRecordUrls, signRecordList };

### Backend (`backend/.env`)
| Variable | Required | Description |

| `NODE_ENV` | Yes | `production` / `development` |
| `FRONTEND_URL` | **Yes** | Comma-separated CORS allowlist, e.g. `https://app.client.com` |
| `CLOUDINARY_CLOUD_NAME` / `CLOUDINARY_API_KEY` / `CLOUDINARY_API_SECRET` | Recommended | Screenshot cloud storage. Without these, screenshots fall back to local disk (served only via signed URLs) |
| `SMTP_HOST` / `SMTP_PORT` / `SMTP_USER` / `SMTP_PASS` / `SMTP_FROM` | Recommended | Without these, emails are SIMULATED (logged, not delivered) and a boot warning is printed |
| `API_PUBLIC_URL` | Recommended | Public URL of the backend (used in emails/installer) |

| Variable | Required | Description |
|---|---|---|
| `VITE_API_URL` | **Yes (production)** | e.g. `https://api.client.com/api`. Defaults to `http://localhost:5000/api` |

### Desktop Agent
Server URL resolution order (first match wins):
1. `API_URL` environment variable
2. `server-config.json` next to the executable (`{ "SERVER_URL": "https://api.client.com" }`)
3. Fallback: `http://localhost:5000`

Set `server-config.json` before building the installer for the client's environment. Invalid URLs are rejected with a warning.

## 2. Database Migrations (baseline strategy)

The project uses a **baseline migration**: `backend/prisma/migrations/000000000000_baseline/migration.sql`
was generated from `schema.prisma` and matches the current production schema.

**Existing database (already has data — e.g. restored from dump):**
```bash
# 1. BACKUP FIRST (see section 3)
# 2. Mark the baseline as applied WITHOUT running it:
npx prisma migrate resolve --applied 000000000000_baseline
# 3. Future schema changes:
npx prisma migrate dev --name <change_name>
```

**Fresh empty database:**
```bash
npx prisma migrate deploy
```

**Rules:** never run `prisma migrate reset` or `db push --force-reset` against real data;
always test migrations on a staging copy first; always take a verified backup before migrating.

## 3. Backup & Recovery

**Backup (MySQL/MariaDB):**
```bash
mysqldump -u <user> -p <database> > backup_$(date +%F).sql
```
Also back up `backend/public/uploads/` if Cloudinary is not configured (local screenshot storage).

**Restore:**
```bash
mysql -u <user> -p <database> < backup_YYYY-MM-DD.sql
```

**Recovery checklist:** restore DB → restore uploads dir → set env vars → `npx prisma generate && npx prisma migrate resolve --applied 000000000000_baseline` → start server → verify `/api/agent/health`.

The agent installer is currently unsigned; Windows SmartScreen will warn users.

1. Obtain a code-signing certificate (OV/EV) from a trusted CA (e.g. DigiCert, Sectigo).

3. Configure `agent/package.json` → electron-builder:
